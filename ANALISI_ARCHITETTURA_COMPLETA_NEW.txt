1. getSheetContext(sheetName)
   SOSTITUISCE: Pattern ripetuto ~50 volte
   
   VECCHIO PATTERN (20+ righe):
     const sheet = SHEETS.get(SHEETS.SHEET_NAMES.Fatture);
     if (!sheet) {
       LOG.error('MY_FUNCTION', 'Foglio non trovato');
       UTIL.showToast('Errore: foglio non trovato', 'Errore', 10);
       return;
     }
     const headerRow = SHEETS._findHeaderRow(sheet, SHEETS.SHEET_NAMES.Fatture);
     const idx = SHEETS.headerIndex(SHEETS.SHEET_NAMES.Fatture);
     if (!idx || Object.keys(idx).length === 0) {
       LOG.error('MY_FUNCTION', 'Schema non trovato');
       return;
     }
     const lastRow = sheet.getLastRow();
     const lastCol = sheet.getLastColumn();
   
   NUOVO PATTERN (2 righe):
     const ctx = SHARED_UTILS.getSheetContext(SHEETS.SHEET_NAMES.Fatture);
     if (!ctx) return;
     // Accesso diretto: ctx.sheet, ctx.headerRow, ctx.idx, ctx.lastRow, ctx.lastCol
   
   RISPARMIO: ~1000 righe (-50 occorrenze × 20 righe)

2. safeExecute(fn, contextName, options)
   SOSTITUISCE: Pattern ripetuto ~60 volte
   
   VECCHIO PATTERN (15+ righe):
     try {
       LOG.info('IMPORT', 'Inizio operazione');
       // ... logica (50 righe)
       LOG.info('IMPORT', 'Completato');
       UTIL.showToast('Operazione completata', 'Successo', 5);
     } catch (e) {
       LOG.error('IMPORT', 'Errore', { error: e.message, stack: e.stack });
       UTIL.showToast('Errore durante operazione', 'Errore', 10);
     }
   
   NUOVO PATTERN (8 righe):
     return SHARED_UTILS.safeExecute(
       () => {
         LOG.info('IMPORT', 'Inizio operazione');
         // ... logica (50 righe)
         return { success: true };
       },
       'IMPORT',
       {
         errorMessage: 'Errore durante operazione',
         onSuccess: () => UTIL.showToast('Operazione completata', 'Successo', 5)
       }
     );
   
   RISPARMIO: ~600 righe (-60 occorrenze × 10 righe)

3. isValidDate(dateValue)
   SOSTITUISCE: Pattern ripetuto ~25 volte

     const dataDoc = row[idx.Data];
